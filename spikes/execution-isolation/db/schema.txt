CREATE FUNCTION core.person() RETURNS text LANGUAGE sql STABLE SECURITY DEFINER SET search_path=pg_catalog AS $$
 SELECT delegated_to FROM core.identities WHERE actor=session_user;
$$;
CREATE FUNCTION core.can_read(target text) RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path=pg_catalog AS $$
 SELECT EXISTS(SELECT 1 FROM core.resources WHERE id=target AND core.person()=ANY(readers));
$$;
ALTER TABLE core.resources ENABLE ROW LEVEL SECURITY;

ALTER TABLE core.proposals ENABLE ROW LEVEL SECURITY;

CREATE FUNCTION core.propose(pid uuid, resource text, parameters jsonb) RETURNS text LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $$
DECLARE checksum text;
BEGIN
 IF NOT core.can_read(resource) THEN RAISE EXCEPTION 'access denied' USING ERRCODE='42501'; END IF;
 IF jsonb_typeof(parameters) <> 'object' OR parameters IS NULL THEN RAISE EXCEPTION 'invalid parameters'; END IF;
 checksum := encode(sha256(convert_to(jsonb_build_object('id',pid,'target',resource,'requester',session_user,'parameters',parameters)::text,'UTF8')),'hex');
 INSERT INTO core.proposals(id,target,requester,delegated_to,params,digest) VALUES(pid,resource,session_user,core.person(),parameters,checksum);
 INSERT INTO core.audit(proposal_id,actor,event,details) VALUES(pid,session_user,'proposed',jsonb_build_object('digest',checksum,'target',resource,'parameters',parameters,'delegated_to',core.person()));
 RETURN checksum;
END $$;
CREATE FUNCTION core.review(pid uuid, checksum text, verdict text) RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $$
DECLARE p core.proposals;
BEGIN
 SELECT * INTO p FROM core.proposals WHERE id=pid FOR UPDATE;
 IF p.id IS NULL OR session_user <> 'lead' OR NOT core.can_read(p.target) OR p.delegated_to=core.person() THEN RAISE EXCEPTION 'access denied' USING ERRCODE='42501'; END IF;
 IF checksum IS DISTINCT FROM p.digest THEN RAISE EXCEPTION 'digest mismatch'; END IF;
 IF verdict NOT IN ('approved','rejected') OR verdict IS NULL THEN RAISE EXCEPTION 'invalid decision'; END IF;
 IF p.decision IS NOT NULL THEN RAISE EXCEPTION 'already reviewed'; END IF;
 UPDATE core.proposals SET decision=verdict,reviewer=session_user WHERE id=pid;
 INSERT INTO core.audit(proposal_id,actor,event,details) VALUES(pid,session_user,verdict,jsonb_build_object('digest',checksum));
END $$;
CREATE FUNCTION core.execute(pid uuid, checksum text, parameters jsonb) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $$
DECLARE p core.proposals; result jsonb;
BEGIN
 SELECT * INTO p FROM core.proposals WHERE id=pid FOR UPDATE;
 IF p.id IS NULL OR p.requester<>session_user OR NOT core.can_read(p.target) THEN RAISE EXCEPTION 'access denied' USING ERRCODE='42501'; END IF;
 IF p.decision IS DISTINCT FROM 'approved' THEN RAISE EXCEPTION 'not approved'; END IF;
 IF checksum IS DISTINCT FROM p.digest OR parameters IS DISTINCT FROM p.params THEN RAISE EXCEPTION 'approved content mismatch'; END IF;
 SELECT params INTO result FROM core.effects WHERE proposal_id=pid;
 IF FOUND THEN RETURN result; END IF;
 -- Efeito controlado no mesmo banco; não representa atomicidade com fornecedor remoto.
 UPDATE core.resources SET payload=p.params WHERE id=p.target;
 INSERT INTO core.effects VALUES(pid,p.target,p.params,current_user,clock_timestamp());
 INSERT INTO core.audit(proposal_id,actor,event,details) VALUES(pid,session_user,'executed',jsonb_build_object('target',p.target,'parameters',p.params,'reviewer',p.reviewer,'credential_role',current_user,'result','applied'));
 RETURN p.params;
END $$;
